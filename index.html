      关于千克起始页

      做一套真正干净、好用、不烦人的工具。

      极简、无广告、开源，这是从一开始就没打算妥协的三件事。

      为什么做？
      因为市面上不是没有同类工具，就是不好用；好用的要付费，免费的不是捆绑就是找不到。与其等一个完美的工具出现，不如自己手搓一个。

      一切的开始，其实很简单——
      嫌弃浏览器主页不够好看，就动手写了一个起始页。
      后来发现，这种“嫌弃”能延伸到很多地方：壁纸、收银、传文件、服务器启动器……
      于是就有了这一堆系列。

      千克起始页不是什么大平台，
      只是一个开发者对自己日常需求的诚实回应。

      如果你也在用，我很感谢你。
      虽然现在可能还没什么人用，但这份感谢是真的。

      贡献者就我一个，
      但背后还有 DeepSeek 和 Trae Work。
      它们不是人，但帮了我很多。

    function initParticles(canvas) {
      const container = canvas.parentElement;
      if (!container) return;
      const dpr = Math.min(window.devicePixelRatio || 1, 2);
      const renderer = new Renderer({ canvas, dpr, depth: false, alpha: true });
      const gl = renderer.gl;
      gl.clearColor(0, 0, 0, 0);

      const camera = new Camera(gl, { fov: 15 });
      camera.position.set(0, 0, 20);

      const resize = () => {
        const w = container.clientWidth || window.innerWidth;
        const h = container.clientHeight || window.innerHeight;
        renderer.setSize(w, h);
        camera.perspective({ aspect: gl.canvas.width / gl.canvas.height });
      };
      window.addEventListener('resize', resize);
      resize();

      const count = 200;
      const positions = new Float32Array(count * 3);
      const randoms = new Float32Array(count * 4);
      const colors = new Float32Array(count * 3);
      for (let i = 0; i < count; i++) {
        let x, y, z, len;
        do {
          x = Math.random() * 2 - 1; y = Math.random() * 2 - 1; z = Math.random() * 2 - 1;
          len = x * x + y * y + z * z;
        } while (len > 1 || len === 0);
        const r = Math.cbrt(Math.random());
        positions.set([x * r, y * r, z * r], i * 3);
        randoms.set([Math.random(), Math.random(), Math.random(), Math.random()], i * 4);
        const col = hexToRgb(defaultColors[Math.floor(Math.random() * defaultColors.length)]);
        colors.set(col, i * 3);
      }
      const geometry = new Geometry(gl, {
        position: { size: 3, data: positions },
        random: { size: 4, data: randoms },
        color: { size: 3, data: colors },
      });
      const program = new Program(gl, {
        vertex, fragment,
        uniforms: {
          uTime: { value: 0 },
          uSpread: { value: 10 },
          uBaseSize: { value: 100 * dpr },
          uSizeRandomness: { value: 1 },
          uAlphaParticles: { value: 0 },
        },
        transparent: true,
        depthTest: false,
      });
      const particles = new Mesh(gl, { mode: gl.POINTS, geometry, program });

      const screen = canvas.closest('.screen-down, .screen-about');
      let raf, lastTime = performance.now(), elapsed = 0;
      const update = t => {
        raf = requestAnimationFrame(update);
        if (!screen || !screen.classList.contains('active')) { lastTime = t; return; }
        const delta = t - lastTime; lastTime = t;
        elapsed += delta * 0.1;
        program.uniforms.uTime.value = elapsed * 0.001;
        particles.rotation.x = Math.sin(elapsed * 0.0002) * 0.1;
        particles.rotation.y = Math.cos(elapsed * 0.0005) * 0.15;
        particles.rotation.z += 0.001;
        renderer.render({ scene: particles, camera });
      };
      raf = requestAnimationFrame(update);
    }

    document.querySelectorAll('.bg-canvas').forEach(initParticles);
  })();
